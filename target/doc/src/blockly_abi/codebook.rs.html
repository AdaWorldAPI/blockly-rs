1//! The Blockly-type → [`FnIndex`] codebook.
2//!
3//! Harvested from the Apache-2.0 Blockly block definitions
4//! (`packages/blockly/blocks/*.ts`) against the `ogar-blockly` palette:
5//! **101 rows over 65 block types** — 96 exact (the palette's own doc comment
6//! names the block), 2 inferred, 3 with **no mapping at all**.
7//!
8//! # Two kinds of dropdown, and the difference matters
9//!
10//! A dropdown field is not always part of the opcode:
11//!
12//! - **[`CodeRole::Selector`]** — the code chooses WHICH function.
13//!   `logic_compare` + `LT` is [`FnIndex::LT`]; the same block with `GT` is a
14//!   different function. The code is consumed by resolution and never becomes
15//!   an immediate.
16//! - **[`CodeRole::ValueParam`]** — the code is an ARGUMENT.
17//!   `math_constant` is always [`FnIndex::CONSTANT`]; whether it yields π or e
18//!   is a value. `text_charAt`'s `WHERE` likewise.
19//!
20//! Collapsing the two would either explode the palette (a slot per constant)
21//! or silently drop the parameter. Both are recorded, and a `ValueParam` whose
22//! byte encoding is not yet defined is **refused** rather than guessed.
23//!
24//! # The three gaps — deliberately unmapped
25//!
26//! These Blockly blocks have no palette entry. They are NOT minted here:
27//! codebook ids are permanent, and a mint is an operator decision with a
28//! ledger entry (OGAR `docs/BLOCK-EDITOR-PLAN.md` § "Standing rules"). Casting
29//! one yields [`CastError::UnknownOpcode`](crate::CastError::UnknownOpcode).
30//!
31//! | block | what it does | why nothing covers it |
32//! |---|---|---|
33//! | `math_on_list[RANDOM]` | random element of a list | `ON_LIST`'s doc enumerates sum/min/max/average/median/mode/std_dev — not random |
34//! | `lists_reverse` | reverse a list | `REVERSE` (`0x6F`) is scoped to `text_reverse`; the `0x70..0x7F` list family has no reverse |
35//! | `lists_getIndex[GET_REMOVE]` | read AND delete in one op | compound; `LIST_GET` reads, `LIST_DELETE` deletes, and giving `LIST_GET` a silent side effect would misrepresent the block |
36//!
37//! # Palette entries no Blockly block reaches
38//!
39//! `FOREVER`, `WAIT`, `WAIT_UNTIL`, `STOP` are Scratch-only by design.
40//! `CONTAINS` and `LIST_CONTAINS` have doc comments citing `text_contains` /
41//! `lists_contains` — **neither block type exists in Blockly**; the real
42//! sources are Scratch's `operator_contains` / `data_listcontainsitem`, so the
43//! `-shaped` phrasing in those doc comments is hedging, not a Blockly citation.
44//! `PROC_ARG` has no dedicated block: argument reads go through an ordinary
45//! `variables_get` bound to the argument's variable model.
46

49/// What a block's dropdown code does during resolution.
50#[derive(Debug, Clone, Copy, PartialEq, Eq)]
51pub enum CodeRole {
52    /// The block has no dropdown that participates in resolution.
53    None,
54    /// The code chooses WHICH function — consumed by resolution.
55    Selector,
56    /// The code is an ARGUMENT to a fixed function — must become a value.
57    ValueParam,
58}

60/// One resolved block type: its function, and what its dropdown code meant.
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62pub struct OpcodeMapping {
63    /// The function this block invokes.
64    pub function: FnIndex,
65    /// Whether the dropdown code selected the function or is an argument.
66    pub role: CodeRole,
67}

69/// Resolve a Blockly block type (with its dropdown code, if any) to a mapping.
70///
71/// Returns `None` for the three deliberate gaps and for any unknown type —
72/// never a guess.
73#[must_use]
74pub fn resolve(ty: &str, code: Option<&str>) -> Option<OpcodeMapping> {
75    use CodeRole::{None as NoCode, Selector, ValueParam};
76    let m = |function, role| Some(OpcodeMapping { function, role });
77
78    match (ty, code) {
79        // ── logic ───────────────────────────────────────────────────────
80        ("logic_boolean", Some("TRUE")) => m(FnIndex::TRUE, Selector),
81        ("logic_boolean", Some("FALSE")) => m(FnIndex::FALSE, Selector),
82        ("controls_if", _) => m(FnIndex::IF, NoCode),
83        ("controls_ifelse", _) => m(FnIndex::IF_ELSE, NoCode),
84        ("logic_compare", Some("EQ")) => m(FnIndex::EQ, Selector),
85        ("logic_compare", Some("NEQ")) => m(FnIndex::NEQ, Selector),
86        ("logic_compare", Some("LT")) => m(FnIndex::LT, Selector),
87        ("logic_compare", Some("LTE")) => m(FnIndex::LTE, Selector),
88        ("logic_compare", Some("GT")) => m(FnIndex::GT, Selector),
89        ("logic_compare", Some("GTE")) => m(FnIndex::GTE, Selector),
90        ("logic_operation", Some("AND")) => m(FnIndex::AND, Selector),
91        ("logic_operation", Some("OR")) => m(FnIndex::OR, Selector),
92        ("logic_negate", _) => m(FnIndex::NOT, NoCode),
93        ("logic_null", _) => m(FnIndex::NULL, NoCode),
94        ("logic_ternary", _) => m(FnIndex::TERNARY, NoCode),
95
96        // ── loops ───────────────────────────────────────────────────────
97        ("controls_repeat" | "controls_repeat_ext", _) => m(FnIndex::REPEAT, NoCode),
98        ("controls_whileUntil", Some("WHILE")) => m(FnIndex::WHILE, Selector),
99        ("controls_whileUntil", Some("UNTIL")) => m(FnIndex::REPEAT_UNTIL, Selector),
100        ("controls_for", _) => m(FnIndex::FOR_RANGE, NoCode),
101        ("controls_forEach", _) => m(FnIndex::FOR_EACH, NoCode),
102        ("controls_flow_statements", Some("BREAK")) => m(FnIndex::BREAK, Selector),
103        ("controls_flow_statements", Some("CONTINUE")) => m(FnIndex::CONTINUE, Selector),
104
105        // ── math ────────────────────────────────────────────────────────
106        ("math_number", _) => m(FnIndex::NUMBER, NoCode),
107        ("math_arithmetic", Some("ADD")) => m(FnIndex::ADD, Selector),
108        ("math_arithmetic", Some("MINUS")) => m(FnIndex::SUB, Selector),
109        ("math_arithmetic", Some("MULTIPLY")) => m(FnIndex::MUL, Selector),
110        ("math_arithmetic", Some("DIVIDE")) => m(FnIndex::DIV, Selector),
111        ("math_arithmetic", Some("POWER")) => m(FnIndex::POW, Selector),
112        ("math_single", Some("ROOT")) => m(FnIndex::SQRT, Selector),
113        ("math_single", Some("ABS")) => m(FnIndex::ABS, Selector),
114        ("math_single", Some("NEG")) => m(FnIndex::NEG, Selector),
115        ("math_single", Some("LN")) => m(FnIndex::LN, Selector),
116        ("math_single", Some("LOG10")) => m(FnIndex::LOG10, Selector),
117        ("math_single", Some("EXP")) => m(FnIndex::EXP_E, Selector),
118        ("math_single", Some("POW10")) => m(FnIndex::EXP_10, Selector),
119        ("math_trig", Some("SIN")) => m(FnIndex::SIN, Selector),
120        ("math_trig", Some("COS")) => m(FnIndex::COS, Selector),
121        ("math_trig", Some("TAN")) => m(FnIndex::TAN, Selector),
122        ("math_trig", Some("ASIN")) => m(FnIndex::ASIN, Selector),
123        ("math_trig", Some("ACOS")) => m(FnIndex::ACOS, Selector),
124        ("math_trig", Some("ATAN")) => m(FnIndex::ATAN, Selector),
125        ("math_constant", _) => m(FnIndex::CONSTANT, ValueParam),
126        ("math_number_property", _) => m(FnIndex::NUMBER_PROPERTY, ValueParam),
127        ("math_change", _) => m(FnIndex::VAR_CHANGE, NoCode),
128        ("math_round", Some("ROUND")) => m(FnIndex::ROUND, Selector),
129        ("math_round", Some("ROUNDUP")) => m(FnIndex::CEIL, Selector),
130        ("math_round", Some("ROUNDDOWN")) => m(FnIndex::FLOOR, Selector),
131        // GAP: math_on_list[RANDOM] — ON_LIST does not enumerate `random`.
132        ("math_on_list", Some("RANDOM")) => Option::None,
133        ("math_on_list", _) => m(FnIndex::ON_LIST, ValueParam),
134        ("math_modulo", _) => m(FnIndex::MOD, NoCode),
135        ("math_constrain", _) => m(FnIndex::CONSTRAIN, NoCode),
136        ("math_random_int", _) => m(FnIndex::RANDOM_INT, NoCode),
137        ("math_random_float", _) => m(FnIndex::RANDOM_FLOAT, NoCode),
138        ("math_atan2", _) => m(FnIndex::ATAN2, NoCode),
139
140        // ── text ────────────────────────────────────────────────────────
141        ("text", _) => m(FnIndex::TEXT, NoCode),
142        ("text_join", _) => m(FnIndex::JOIN, NoCode),
143        ("text_append", _) => m(FnIndex::APPEND, NoCode),
144        ("text_length", _) => m(FnIndex::LENGTH, NoCode),
145        ("text_isEmpty", _) => m(FnIndex::IS_EMPTY, NoCode),
146        ("text_indexOf", _) => m(FnIndex::INDEX_OF, ValueParam),
147        ("text_charAt", _) => m(FnIndex::CHAR_AT, ValueParam),
148        ("text_getSubstring", _) => m(FnIndex::SUBSTRING, ValueParam),
149        ("text_changeCase", _) => m(FnIndex::CHANGE_CASE, ValueParam),
150        ("text_trim", _) => m(FnIndex::TRIM, ValueParam),
151        ("text_print", _) => m(FnIndex::PRINT, NoCode),
152        ("text_prompt" | "text_prompt_ext", _) => m(FnIndex::PROMPT, ValueParam),
153        ("text_count", _) => m(FnIndex::COUNT, NoCode),
154        ("text_replace", _) => m(FnIndex::REPLACE, NoCode),
155        ("text_reverse", _) => m(FnIndex::REVERSE, NoCode),
156
157        // ── lists ───────────────────────────────────────────────────────
158        ("lists_create_empty", _) => m(FnIndex::LIST_EMPTY, NoCode),
159        ("lists_create_with", _) => m(FnIndex::LIST_WITH, NoCode),
160        ("lists_repeat", _) => m(FnIndex::LIST_REPEAT, NoCode),
161        ("lists_length", _) => m(FnIndex::LIST_LENGTH, NoCode),
162        ("lists_isEmpty", _) => m(FnIndex::LIST_IS_EMPTY, NoCode),
163        ("lists_indexOf", _) => m(FnIndex::LIST_INDEX_OF, ValueParam),
164        ("lists_getIndex", Some("GET")) => m(FnIndex::LIST_GET, Selector),
165        // GAP: GET_REMOVE is compound (read AND delete); neither half covers it.
166        ("lists_getIndex", Some("GET_REMOVE")) => Option::None,
167        ("lists_getIndex", Some("REMOVE")) => m(FnIndex::LIST_DELETE, Selector),
168        ("lists_setIndex", Some("SET")) => m(FnIndex::LIST_SET, Selector),
169        ("lists_setIndex", Some("INSERT")) => m(FnIndex::LIST_INSERT, Selector),
170        ("lists_getSublist", _) => m(FnIndex::LIST_SUBLIST, ValueParam),
171        ("lists_sort", _) => m(FnIndex::LIST_SORT, ValueParam),
172        ("lists_split", _) => m(FnIndex::LIST_SPLIT, ValueParam),
173        // GAP: lists_reverse — REVERSE (0x6F) is text-scoped; no list reverse.
174        ("lists_reverse", _) => Option::None,
175
176        // ── variables ───────────────────────────────────────────────────
177        ("variables_get" | "variables_get_dynamic", _) => m(FnIndex::VAR_GET, NoCode),
178        ("variables_set" | "variables_set_dynamic", _) => m(FnIndex::VAR_SET, NoCode),
179
180        // ── procedures ──────────────────────────────────────────────────
181        ("procedures_defnoreturn" | "procedures_defreturn", _) => m(FnIndex::PROC_DEF, NoCode),
182        ("procedures_callnoreturn" | "procedures_callreturn", _) => m(FnIndex::PROC_CALL, NoCode),
183        ("procedures_ifreturn", _) => m(FnIndex::RETURN, NoCode),
184
185        _ => Option::None,
186    }
187}

197/// The ordered parameter codes for one `(block type, field name)` dropdown
198/// whose role is [`CodeRole::ValueParam`].
199///
200/// # The encoding, and why the ordinal is OURS
201///
202/// A `ValueParam` byte is the code's **ordinal in this table** — `PI` is `0`,
203/// `E` is `1`, and so on. The block type is already named by the call's
204/// [`FnIndex`], so the ordinal need only be unique *within* the block: no
205/// global parameter namespace is minted, and no palette slot is spent.
206///
207/// The ordinal is deliberately anchored **here** rather than to "whatever
208/// order Blockly's array has today". Blockly reordering an options array is a
209/// cosmetic upstream change; if the encoding tracked it, that cosmetic change
210/// would silently reinterpret every stored program — `math_constant` `0` would
211/// stop meaning π. Anchoring the ordinal in this table converts that hazard
212/// into a **loud** one: upstream reordering then breaks the drift test
213/// (`the_value_param_option_sets_are_pinned`) and nothing else. Permanence lives with the codebook, which is the same reason
214/// the three gaps are refused rather than minted.
215///
216/// Widths, measured against the Apache-2.0 definitions (largest set is 8):
217/// every set fits a byte with three orders of magnitude to spare, so the
218/// `u8` is not a squeeze.
219///
220/// Returns `None` for any pair this codebook does not name — never a guess.
221#[must_use]
222pub fn value_param_codes(ty: &str, field: &str) -> Option<&'static [&'static str]> {
223    Some(match (ty, field) {
224        ("math_constant", "CONSTANT") => {
225            &["PI", "E", "GOLDEN_RATIO", "SQRT2", "SQRT1_2", "INFINITY"]
226        }
227        ("math_number_property", "PROPERTY") => &[
228            "EVEN",
229            "ODD",
230            "PRIME",
231            "WHOLE",
232            "POSITIVE",
233            "NEGATIVE",
234            "DIVISIBLE_BY",
235        ],
236        // RANDOM is absent BY CONSTRUCTION: it is one of the three gaps, and
237        // `resolve` already refuses it. Listing it here would hand a byte to a
238        // block the palette cannot name.
239        ("math_on_list", "OP") => &["SUM", "MIN", "MAX", "AVERAGE", "MEDIAN", "MODE", "STD_DEV"],
240        ("text_indexOf" | "lists_indexOf", "END") => &["FIRST", "LAST"],
241        ("text_charAt", "WHERE") => &["FROM_START", "FROM_END", "FIRST", "LAST", "RANDOM"],
242        // Two dropdowns, and their sets genuinely DIFFER in the third slot
243        // (FIRST vs LAST) — which is why the table keys on the field name and
244        // not the block type alone.
245        ("text_getSubstring" | "lists_getSublist", "WHERE1") => {
246            &["FROM_START", "FROM_END", "FIRST"]
247        }
248        ("text_getSubstring" | "lists_getSublist", "WHERE2") => &["FROM_START", "FROM_END", "LAST"],
249        ("text_changeCase", "CASE") => &["UPPERCASE", "LOWERCASE", "TITLECASE"],
250        ("text_trim", "MODE") => &["BOTH", "LEFT", "RIGHT"],
251        ("text_prompt" | "text_prompt_ext", "TYPE") => &["TEXT", "NUMBER"],
252        ("lists_sort", "TYPE") => &["NUMERIC", "TEXT", "IGNORE_CASE"],
253        // The codes really are the strings "1" and "-1" in the source. They
254        // are NOT read as numbers: `-1` would not survive a `u8`, and the
255        // ordinal encoding sidesteps the question entirely.
256        ("lists_sort", "DIRECTION") => &["1", "-1"],
257        // JUDGMENT, recorded rather than silently taken: SPLIT (string → list)
258        // and JOIN (list → string) are arguably different *operations*, which
259        // would make this a Selector. The palette mints ONE slot
260        // (`LIST_SPLIT`), so treating the code as a parameter is what that
261        // slot actually supports, and it is lossless — the byte survives.
262        // Promoting it to two slots is a mint, i.e. an operator decision.
263        ("lists_split", "MODE") => &["SPLIT", "JOIN"],
264        _ => return None,
265    })
266}

268/// Encode a `ValueParam` dropdown code as its immediate byte.
269///
270/// Returns `None` if the codebook does not name this `(type, field)` pair, or
271/// names it but not this code — the caller refuses rather than substituting a
272/// default, because a wrong parameter byte is a program that runs and computes
273/// the wrong thing.
274#[must_use]
275pub fn encode_value_param(ty: &str, field: &str, code: &str) -> Option<u8> {
276    let codes = value_param_codes(ty, field)?;
277    let idx = codes.iter().position(|c| *c == code)?;
278    // The largest set is 8; the pin below makes this unreachable in practice,
279    // and the fallible conversion means a future oversized set refuses rather
280    // than wrapping to a valid-looking byte.
281    u8::try_from(idx).ok()
282}
283
284/// The inverse of [`encode_value_param`] — the code a stored byte denotes.
285#[must_use]
286pub fn decode_value_param(ty: &str, field: &str, byte: u8) -> Option<&'static str> {
287    value_param_codes(ty, field)?
288        .get(usize::from(byte))
289        .copied()
290}

296    #[test]
297    fn a_dropdown_selector_picks_different_functions() {
298        // The whole point of Selector: the SAME block type with different
299        // codes must land on DIFFERENT palette slots.
300        let lt = resolve("logic_compare", Some("LT")).unwrap();
301        let gt = resolve("logic_compare", Some("GT")).unwrap();
302        assert_eq!(lt.function, FnIndex::LT);
303        assert_eq!(gt.function, FnIndex::GT);

561    #[test]
562    fn a_value_param_code_encodes_to_its_ordinal_and_back() {
563        // Can-fire: distinct codes must reach DISTINCT bytes, or the whole
564        // encoding is decoration and math_constant[PI] == math_constant[E].
565        let pi = encode_value_param("math_constant", "CONSTANT", "PI").unwrap();
566        let e = encode_value_param("math_constant", "CONSTANT", "E").unwrap();
567        assert_eq!(pi, 0);
568        assert_eq!(e, 1);
569        assert_ne!(pi, e);
570        // Round-trip every code of every set, so a table edit that breaks the
571        // inverse is caught here and not in a stored program.
572        for (ty, field) in [
573            ("math_constant", "CONSTANT"),
574            ("math_number_property", "PROPERTY"),
575            ("math_on_list", "OP"),
576            ("text_charAt", "WHERE"),
577            ("lists_sort", "DIRECTION"),
578            ("lists_split", "MODE"),
579        ] {
580            for code in value_param_codes(ty, field).unwrap() {
581                let byte = encode_value_param(ty, field, code).unwrap();
582                assert_eq!(decode_value_param(ty, field, byte), Some(*code));
583            }
584        }
585    }

588    fn the_two_dropdowns_of_one_block_do_not_share_a_table() {
589        // WHERE1 ends FIRST, WHERE2 ends LAST. Keying on the block type alone
590        // would make ordinal 2 mean two different things on one block — and
591        // would pass a same-table test vacuously, since the first two entries
592        // ARE identical.
593        assert_eq!(
594            encode_value_param("text_getSubstring", "WHERE1", "FROM_END"),
595            encode_value_param("text_getSubstring", "WHERE2", "FROM_END")
596        );
597        assert_eq!(
598            decode_value_param("text_getSubstring", "WHERE1", 2),
599            Some("FIRST")
600        );
601        assert_eq!(
602            decode_value_param("text_getSubstring", "WHERE2", 2),
603            Some("LAST")
604        );
605        // FIRST is not reachable at all on WHERE2, and LAST not on WHERE1.
606        assert_eq!(
607            encode_value_param("text_getSubstring", "WHERE2", "FIRST"),
608            None
609        );
610        assert_eq!(
611            encode_value_param("text_getSubstring", "WHERE1", "LAST"),
612            None
613        );

617    fn an_unknown_value_param_is_refused_not_defaulted() {
618        // Silence twin for the encoding test: a code the codebook does not

635    /// The D3 palette drift anchor for the FUNCTION half.
636    ///
637    /// The census test above compares against `FnIndex::LT` — the *symbol*. If
638    /// `ogar-blockly` renumbered `LT` from `0x32` to `0x33`, every symbolic
639    /// assertion in this file would still pass while every stored program
640    /// silently changed meaning. This pins the BYTES.
641    #[test]
642    fn the_palette_byte_values_are_pinned() {
643        let pinned: &[(&str, Option<&str>, u8)] = &[
644            ("controls_if", None, 0x01),
645            ("controls_repeat", None, 0x03),
646            ("controls_whileUntil", Some("WHILE"), 0x05),
647            ("logic_negate", None, 0x22),
648            ("logic_compare", Some("EQ"), 0x30),
649            ("logic_compare", Some("LT"), 0x32),
650            ("logic_compare", Some("GT"), 0x34),
651            ("math_arithmetic", Some("ADD"), 0x40),
652            ("math_number", None, 0x46),
653            ("math_single", Some("ABS"), 0x47),
654            ("math_trig", Some("SIN"), 0x51),
655            ("math_constant", Some("PI"), 0x5C),
656            ("text", None, 0x60),
657            ("lists_getIndex", Some("GET"), 0x76),
658            ("variables_get", None, 0x80),
659            ("procedures_callnoreturn", None, 0x84),
660        ];
661        for (ty, code, byte) in pinned {
662            let got = resolve_opcode(ty, *code)
663                .unwrap_or_else(|| panic!("{ty} lost its mapping"))
664                .0;
665            assert_eq!(got, *byte, "{ty} moved off palette byte {byte:#04x}");
666        }
667        // Anti-vacuity: the pins must be DISTINCT bytes. A palette that
668        // collapsed every entry onto one slot would satisfy a per-row check
669        // only if the expected bytes were also collapsed — this catches the
670        // case where the table itself is edited to match a broken palette.
671        let mut bytes: Vec<u8> = pinned.iter().map(|(_, _, b)| *b).collect();
672        bytes.sort_unstable();
673        let before = bytes.len();
674        bytes.dedup();
675        assert_eq!(bytes.len(), before, "two pins share a palette byte");
676
677        // Nothing the Blockly frontend reaches may land in the DEVICE family
678        // (`0x90..`), which is reserved for hardware/robotics blocks that have
679        // no Blockly-core counterpart.
680        for (ty, _, byte) in pinned {
681            assert!(
682                *byte < ogar_blockly::DEVICE_FAMILY_FLOOR,
683                "{ty} reached the reserved device family"
684            );
685        }
686        // …and the floor is a real boundary, not a vacuous one: it must sit
687        // ABOVE every pin (or the loop above would be impossible) and BELOW
688        // the top of the byte range (or "below the floor" would exclude
689        // nothing and the loop would be free).
690        const { assert!(ogar_blockly::DEVICE_FAMILY_FLOOR < u8::MAX) };
691        let highest_pin = pinned.iter().map(|(_, _, b)| *b).max().unwrap();
692        assert!(highest_pin < ogar_blockly::DEVICE_FAMILY_FLOOR);
693    }
