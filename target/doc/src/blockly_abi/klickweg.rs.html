1//! Klickwege addressing — clicking a block is an **address**, never a handler.
2//!
3//! # The trap this exists to avoid (a2ui charter T2)
4//!
5//! *"Behavior never rides the surface."* An `onClick: <lambda>` in a component
6//! tree is the same hijack as `DEFINE EVENT … WHEN … THEN …` in DDL: it puts
7//! behaviour on the transport. The a2ui charter's answer is that a click
8//! travels as `ActionInvoke { key, action_ordinal, args }` — an **ordinal into
9//! the class's action set** — and the behaviour is a property of the Core node
10//! the address resolves to.
11//!
12//! This module is the block-editor half of that: given a workspace and the id
13//! of the block a user clicked, produce the address. It never produces a
14//! callback, a closure, or a script fragment, and it cannot — nothing here
15//! returns anything invocable.
16//!
17//! # Why the ordinal is the CALL INDEX and not a block counter
18//!
19//! The ordinal has to name something that survives the trip. A block id is
20//! editor state; a position is presentation; a "nth block in the XML" ordering
21//! is neither, because Blockly's own tree order is not the program's order.
22//!
23//! The call index is the one number that is already the program: it is where
24//! [`lower_script`](crate::lower_script) put that block's
25//! [`Call`](ogar_blockly::Call) in the body, so
26//! `raise_calls(body)[ordinal]` IS the clicked block's function, by
27//! construction rather than by convention. That makes the address checkable
28//! against the ABI instead of merely consistent with itself — and it is
29//! exactly what the falsifier
30//! (`an_ordinal_indexes_the_clicked_blocks_own_call`) asserts.
31//!
32//! Consequence worth stating: the traversal here MUST match the cast's
33//! post-order. It is not "the same idea implemented twice" — it walks the same
34//! shape for the same reason, and the test compares the two rather than
35//! trusting them to agree.
36//!
37//! # What a consumer does with this
38//!
39//! `a2ui-server`'s `KlickwegEdge` wants `{from_key, class_id, ordinal,
40//! predicate, seq}`. Three of those come from here ([`BlockAddress`]);
41//! `from_key` is the function node's own GUID and `seq` is the session's
42//! monotonic counter — both owned by the session, not by the cast. Nothing in
43//! this crate depends on `a2ui-server`: the block editor produces addresses,
44//! the desktop consumes them, and neither imports the other.
45
46use ogar_blockly::{BlockConcept, LaneShape};
47
48use crate::{BlockRecord, codebook};

50/// Where a click lands, in the ABI's own coordinates.
51#[derive(Debug, Clone, PartialEq, Eq)]
52pub struct BlockAddress {
53    /// The V3 render classid — canon-high, `concept << 16 | app_prefix`.
54    pub class_id: u32,
55    /// The clicked block's **call index** in the lowered body. This is the
56    /// address; `raise_calls(body)[ordinal]` is its call.
57    pub ordinal: u32,
58    /// The edge label — the Blockly block type, with its dropdown code when
59    /// the code participates in resolution.
60    ///
61    /// A label, not a lookup key: the ordinal is what resolves, and the
62    /// predicate is what a human reads in a trace. Deriving it from the block
63    /// type keeps it honest without minting a second vocabulary (charter T1).
64    pub predicate: String,
65}
66
67/// The Klickwege edge label for a block: its type, plus the code when the code
68/// selected the function.
69///
70/// A `ValueParam` code is deliberately NOT appended — it is an argument, and
71/// putting an argument in the edge label would make two clicks on the same
72/// action read as two different actions.
73#[must_use]
74pub fn predicate_for(block: &BlockRecord) -> String {
75    let code = block.dropdown_code();
76    match (code, codebook::resolve(&block.ty, code)) {
77        (Some(c), Some(m)) if m.role == codebook::CodeRole::Selector => {
78            format!("{}[{c}]", block.ty)
79        }
80        _ => block.ty.clone(),
81    }
82}
83
84/// Every block's address, in call order.
85///
86/// The returned vector is indexed BY ordinal: element `i` is the block whose
87/// call sits at index `i`. Walking post-order — operands before their
88/// operator, then `next` — because that is what the cast emits.
89#[must_use]
90pub fn addresses(top: &BlockRecord, app_prefix: u16) -> Vec<(String, BlockAddress)> {
91    let class_id = BlockConcept::Content.render_classid(app_prefix);
92    let mut out = Vec::new();
93    walk_chain(top, class_id, &mut out);
94    out
95}
96
97fn walk_chain(block: &BlockRecord, class_id: u32, out: &mut Vec<(String, BlockAddress)>) {
98    walk_block(block, class_id, out);
99    if let Some(next) = &block.next {
100        walk_chain(next, class_id, out);
101    }
102}
103
104fn walk_block(block: &BlockRecord, class_id: u32, out: &mut Vec<(String, BlockAddress)>) {
105    for (_, operand) in &block.inputs {
106        walk_block(operand, class_id, out);
107    }
108    // `out.len()` before the push IS the call index, because the cast pushes
109    // its call at exactly this point in the same walk.
110    let ordinal = u32::try_from(out.len()).unwrap_or(u32::MAX);
111    out.push((
112        block.id.clone(),
113        BlockAddress {
114            class_id,
115            ordinal,
116            predicate: predicate_for(block),
117        },
118    ));
119}

121/// The address of one clicked block, by id.
122///
123/// Returns `None` for an id the script does not contain — a click on a block
124/// that is not in this function is refused rather than resolved to ordinal 0,
125/// which would fire the wrong action.
126#[must_use]
127pub fn address_of(top: &BlockRecord, block_id: &str, app_prefix: u16) -> Option<BlockAddress> {
128    addresses(top, app_prefix)
129        .into_iter()
130        .find(|(id, _)| id == block_id)
131        .map(|(_, a)| a)
132}
133
134/// Whether a shape can address every block in the script — i.e. the program
135/// fits, so every ordinal names a real call.
136///
137/// An address past the body's capacity is not an address; it is a number that
138/// resolves to nothing. Callers check this before handing addresses out.
139#[must_use]
140pub fn addressable(top: &BlockRecord, shape: LaneShape) -> bool {
141    top.block_count() <= shape.calls_per_function()
142}

164    #[test]
165    fn an_ordinal_indexes_the_clicked_blocks_own_call() {
166        // THE falsifier. The address is only an address if it lands on the
167        // right call — so it is checked against the ABI, not against itself.
168        // A pre-order walk here (parent before operands) would still produce
169        // unique, plausible-looking ordinals and would fail this.
170        let script = five_plus_three();
171        let body = lower_script(LaneShape::Pairs, &script).unwrap();
172        let calls = raise_calls(&body);
173
174        for (id, addr) in addresses(&script, APP) {
175            let call = &calls[addr.ordinal as usize];
176            let expected = match id.as_str() {
177                "a" | "b" => ogar_blockly::FnIndex::NUMBER,
178                "root" => ogar_blockly::FnIndex::ADD,
179                other => panic!("unexpected block {other}"),
180            };
181            assert_eq!(
182                call.function, expected,
183                "block {id} addressed the wrong call"
184            );
185        }
186
187        // Anti-vacuity: the operands must NOT be at the same ordinal as their
188        // operator, or "every address is right" would be trivially satisfiable
189        // by a table of zeroes.
190        let a = address_of(&script, "a", APP).unwrap().ordinal;
191        let b = address_of(&script, "b", APP).unwrap().ordinal;
192        let root = address_of(&script, "root", APP).unwrap().ordinal;
193        assert_eq!((a, b, root), (0, 1, 2));
194        // …and it really is post-order: the operator comes LAST.
195        assert!(root > a && root > b);
196    }
197
198    #[test]
199    fn the_ordinals_are_dense_and_agree_with_the_call_count() {
200        let script = five_plus_three();
201        let addrs = addresses(&script, APP);
202        assert_eq!(addrs.len(), script.block_count());
203        for (i, (_, a)) in addrs.iter().enumerate() {
204            assert_eq!(a.ordinal as usize, i, "ordinals must be dense");
205        }
206        let body = lower_script(LaneShape::Pairs, &script).unwrap();
207        assert_eq!(addrs.len(), body.len());
208    }

210    #[test]
211    fn a_statement_chain_addresses_after_its_operands() {
212        // `next` is the statement sequence and is walked AFTER the block's own
213        // call — so a two-statement script must not interleave.
214        let script = BlockRecord::leaf("text_print", "p1")
215            .with_input(
216                "TEXT",
217                BlockRecord::leaf("math_number", "n1").with_field("NUM", FieldValue::Byte(1)),
218            )
219            .with_next(BlockRecord::leaf("text_print", "p2").with_input(
220                "TEXT",
221                BlockRecord::leaf("math_number", "n2").with_field("NUM", FieldValue::Byte(2)),
222            ));
223        let ord = |id: &str| address_of(&script, id, APP).unwrap().ordinal;
224        assert_eq!((ord("n1"), ord("p1"), ord("n2"), ord("p2")), (0, 1, 2, 3));
225
226        // Cross-checked against the ABI, so this is not just self-consistent.
227        let calls = raise_calls(&lower_script(LaneShape::Pairs, &script).unwrap());
228        assert_eq!(calls[ord("n1") as usize].values[0], 1);
229        assert_eq!(calls[ord("n2") as usize].values[0], 2);
230    }

241    #[test]
242    fn the_classid_is_canon_high_under_the_app_prefix() {
243        let addr = address_of(&five_plus_three(), "root", APP).unwrap();
244        assert_eq!(addr.class_id, 0x1701_1000);
245        assert_eq!(addr.class_id >> 16, 0x1701, "concept must be the HIGH half");
246        assert_eq!(addr.class_id & 0xFFFF, u32::from(APP));
247        // A different app renders the same concept differently — proving the
248        // prefix is live and not baked.
249        let other = address_of(&five_plus_three(), "root", 0x2000).unwrap();
250        assert_ne!(other.class_id, addr.class_id);
251        assert_eq!(other.class_id >> 16, addr.class_id >> 16);
252    }

254    #[test]
255    fn a_selector_code_labels_the_edge_but_an_argument_code_does_not() {
256        // Two-sided on the one judgment this module makes. A Selector code IS
257        // part of which action fired; a ValueParam code is an argument, and
258        // putting it in the label would make two clicks on one action read as
259        // two different actions.
260        let add = BlockRecord::leaf("math_arithmetic", "x")
261            .with_field("OP", FieldValue::Code("ADD".into()));
262        assert_eq!(predicate_for(&add), "math_arithmetic[ADD]");
263
264        let pi = BlockRecord::leaf("math_constant", "y")
265            .with_field("CONSTANT", FieldValue::Code("PI".into()));
266        let e = BlockRecord::leaf("math_constant", "y")
267            .with_field("CONSTANT", FieldValue::Code("E".into()));
268        assert_eq!(predicate_for(&pi), "math_constant");
269        assert_eq!(
270            predicate_for(&pi),
271            predicate_for(&e),
272            "an argument must not change the edge label"
273        );
274        // …and the Selector case really does discriminate, or the first
275        // assertion above would pass for a function that ignores codes.
276        let sub = BlockRecord::leaf("math_arithmetic", "x")
277            .with_field("OP", FieldValue::Code("MINUS".into()));
278        assert_ne!(predicate_for(&add), predicate_for(&sub));
279    }

281    #[test]
282    fn addressability_tracks_the_shapes_real_budget() {
283        let script = five_plus_three();
284        assert!(addressable(&script, LaneShape::Pairs));
285        // `with_next` REPLACES the successor, so a chain of length n is built
286        // from the tail inward — appending in a forward loop would silently
287        // yield a 2-block script and make this test vacuous.
288        let chain = |n: usize| {
289            let mut b = BlockRecord::leaf("text_print", format!("p{}", n - 1));
290            for i in (0..n - 1).rev() {
291                b = BlockRecord::leaf("text_print", format!("p{i}")).with_next(b);
292            }
293            assert_eq!(b.block_count(), n, "the fixture must really be n blocks");
294            b
295        };
296
297        // A program larger than the shape's budget is NOT addressable — every
298        // ordinal past capacity resolves to nothing.
299        let quads = LaneShape::Quads.calls_per_function();
300        assert!(!addressable(&chain(quads + 1), LaneShape::Quads));
301        // Exactly at capacity still is — so the bound is `<=`, not `<`.
302        assert!(addressable(&chain(quads), LaneShape::Quads));
303        // Two-sided across shapes: the wider budget admits what the narrower
304        // one refuses.
305        assert!(addressable(&chain(quads + 1), LaneShape::Pairs));
306    }
